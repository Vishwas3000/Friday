cmake_minimum_required(VERSION 3.22.1)
# Device tests use the debug APK; its LLM kernels still need normal optimized code.
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set_property(GLOBAL PROPERTY JOB_POOLS writer_compile=3)
set(CMAKE_JOB_POOL_COMPILE writer_compile)
project(friday C CXX)
# Friday's own native helpers (research/19): thread placement for Laya's passes.
add_library(friday SHARED affinity.c)

# Source and models are fetched explicitly, with checksums, by scripts/setup_writer.py.
set(LLAMA_SRC "${CMAKE_CURRENT_SOURCE_DIR}/../../../../third_party/llama.cpp")
if(NOT EXISTS "${LLAMA_SRC}/include/llama.h")
    message(FATAL_ERROR "Writer runtime missing. Run: python3 scripts/setup_writer.py --runtime-only")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
# Pixel 8 / Tensor G3. Keep the same supported device scope as the current arm64 build.
set(GGML_CPU_ARM_ARCH "armv8.6-a+dotprod+i8mm+fp16" CACHE STRING "" FORCE)
add_subdirectory("${LLAMA_SRC}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)
add_library(writer SHARED writer_jni.cpp)
target_compile_features(writer PRIVATE cxx_std_17)
target_link_libraries(writer PRIVATE llama android log)
